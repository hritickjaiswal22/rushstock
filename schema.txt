users {
    id (pkey)
    email (unique)
    name 
    password
    role ENUM ('USER', 'ADMIN') DEFAULT("USER")
    created_at
    updated_at
}

products {
    id (pkey)
    title
    description
    price (> 0)
    created_at
    updated_at
}

sales {
    id (pkey)
    author_id
    product_id
    start_at
    end_at
    stock_quantity (>= 0)
    unit_price (> 0)
    created_at
    updated_at

    Index on (start_at,end_at)
    Index on (product_id,start_at,end_at)
    Unique (product_id,start_at,end_at)
    end_at > start_at
    no overlap gist on product_id,start_at,end_at
}

orders {
    id
    sale_id
    user_id
    idempotency_id (unique)
    quantity (> 0)
    status ENUM(PENDING, EXPIRED, SUCCESS, FAILED - last 3 are terminal status hence cannot be changed) 
    expires_at

    Partial unique index on (sale_id,user_id) AND status IN ('PENDING', 'SUCCESS')
}
